# Chapters Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add chapters as first-class containers in Braidr — replacing the old positional `BraidedChapter` model with explicit `Chapter` entities that scenes are assigned to.

**Architecture:** Chapters are stored in a new `chapters` SQLite table; scenes gain `chapter_id` and `scene_order` columns. Five new IPC channels handle chapter CRUD. The braided view renders chapters as draggable containers with scenes inside (dnd-kit nested sortable). Rails, POV, table, and compile views are updated to respect chapters. No chapters → all views look identical to today.

**Tech Stack:** TypeScript, Electron IPC (better-sqlite3), React (dnd-kit, @dnd-kit/sortable), Vitest

private migrate() {
  const sceneColumns = (
    this.db.prepare('PRAGMA table_info(scenes)').all() as { name: string }[]
  ).map(c => c.name);

  if (!sceneColumns.includes('chapter_id')) {
    this.db.exec(
      'ALTER TABLE scenes ADD COLUMN chapter_id TEXT REFERENCES chapters(id) ON DELETE SET NULL'
    );
  }
  if (!sceneColumns.includes('scene_order')) {
    this.db.exec(
      'ALTER TABLE scenes ADD COLUMN scene_order INTEGER NOT NULL DEFAULT 0'
    );
  }

  // Drop legacy positional chapters table — no data is migrated
  this.db.exec('DROP TABLE IF EXISTS braided_chapters');
}
```

saveChapter(chapter: { id: string; title: string; order: number; description?: string }) {
  this.db.prepare(`
    INSERT INTO chapters (id, title, ord, description)
    VALUES (?, ?, ?, ?)
    ON CONFLICT(id) DO UPDATE SET
      title = excluded.title,
      ord   = excluded.ord,
      description = excluded.description
  `).run(chapter.id, chapter.title, chapter.order, chapter.description ?? null);
}

deleteChapter(id: string) {
  // scenes.chapter_id → NULL via ON DELETE SET NULL FK
  this.db.prepare('DELETE FROM chapters WHERE id = ?').run(id);
}

reorderChapters(orderedIds: string[]) {
  const update = this.db.prepare('UPDATE chapters SET ord = ? WHERE id = ?');
  this.db.transaction(() => {
    orderedIds.forEach((id, idx) => update.run(idx, id));
  })();
}
```

  if (!over || active.id === over.id) return;

  const activeData = active.data.current as { type?: string; chapterId?: string } | undefined;
  const overData = over.data.current as { type?: string; chapterId?: string } | undefined;

  if (activeData?.type === 'chapter') {

    const newIndex = sortedChapters.findIndex(ch => ch.id === over.id);
    if (oldIndex !== -1 && newIndex !== -1 && oldIndex !== newIndex) {

      onReorderChapters(reordered.map(ch => ch.id));
    }
    return;
  }

  if (activeData?.type === 'scene') {
    const activeScene = displayedScenes.find(s => s.id === active.id);
    if (!activeScene) return;

    // Drop onto a chapter header → append to that chapter
    if (overData?.type === 'chapter') {
      const targetChapterId = over.id as string;
      const scenesInTarget = scenesByChapter.get(targetChapterId) || [];
      onAssignSceneToChapter(activeScene.id, targetChapterId, scenesInTarget.length);
      return;

    // Drop onto a scene → reorder within chapter or move between chapters
    if (overData?.type === 'scene') {
      const overScene = displayedScenes.find(s => s.id === over.id);
      if (!overScene) return;

      if (activeScene.chapterId === overScene.chapterId) {
        // Reorder within same chapter
        const chapterScenes = [...(scenesByChapter.get(activeScene.chapterId ?? '') || [])];
        const oldIdx = chapterScenes.findIndex(s => s.id === active.id);
        const newIdx = chapterScenes.findIndex(s => s.id === over.id);
        if (oldIdx !== -1 && newIdx !== -1 && oldIdx !== newIdx) {
          const reordered = arrayMove(chapterScenes, oldIdx, newIdx);
          reordered.forEach((s, idx) => onAssignSceneToChapter(s.id, s.chapterId, idx));
        }
      } else {
        // Move to different chapter — insert before the over scene
        const targetChapterId = overScene.chapterId;
        const targetScenes = [...(scenesByChapter.get(targetChapterId ?? '') || [])];
        const insertIdx = targetScenes.findIndex(s => s.id === over.id);
        const finalIdx = insertIdx === -1 ? targetScenes.length : insertIdx;

        // Move active scene to target chapter
        onAssignSceneToChapter(activeScene.id, targetChapterId, finalIdx);

        // Renumber remaining scenes in the source chapter
        const srcScenes = (scenesByChapter.get(activeScene.chapterId ?? '') || [])
          .filter(s => s.id !== activeScene.id);
        srcScenes.forEach((s, idx) => onAssignSceneToChapter(s.id, s.chapterId, idx));

        // Renumber scenes in target chapter after insertion
        const newTargetScenes = [
          ...targetScenes.slice(0, finalIdx),
          activeScene,
          ...targetScenes.slice(finalIdx),

  const sorted = [...chapters].sort((a, b) => a.order - b.order);
  const result: JSX.Element[] = [];
  sorted.forEach(ch => {
    const chScenes = scenes.filter(s => s.chapterId === ch.id).sort((a, b) => a.sceneOrder - b.sceneOrder);
    if (chScenes.length === 0) return;
    result.push(
      <tr key={`ch-${ch.id}`} className="table-chapter-header">
        <td colSpan={999}>{ch.title}</td>
      </tr>
    );
    chScenes.forEach(s => result.push(<TableRow key={s.id} scene={s} ... />));
  });
  return result;
}
